"""Discover and stage static import_playbook dependencies."""
from pathlib import Path
import shutil
import yaml

PLAYBOOK_ROOT = Path("/opt/ansible/playbooks").resolve()
IMPORT_KEYS = ("import_playbook", "ansible.builtin.import_playbook")


class PlaybookDependencyError(ValueError):
    """A playbook dependency is missing, invalid, or unsafe."""


def _resolve_playbook(path, root):
    try:
        resolved = path.resolve(strict=True)
    except FileNotFoundError as exc:
        raise FileNotFoundError(f"Playbook file not found: {path}") from exc
    if not resolved.is_file():
        raise PlaybookDependencyError(f"Not a regular file: {resolved}")
    try:
        resolved.relative_to(root)
    except ValueError as exc:
        raise PlaybookDependencyError(
            f"Playbook path escapes approved root {root}: {resolved}"
        ) from exc
    if resolved.suffix.lower() not in (".yml", ".yaml"):
        raise PlaybookDependencyError(
            f"Playbook must have a .yml or .yaml extension: {resolved}"
        )
    return resolved


def _read_imports(path):
    try:
        with path.open("r", encoding="utf-8") as stream:
            document = yaml.safe_load(stream)
    except yaml.YAMLError as exc:
        raise PlaybookDependencyError(f"Invalid YAML in {path}: {exc}") from exc
    except OSError as exc:
        raise PlaybookDependencyError(f"Cannot read playbook {path}: {exc}") from exc

    if document is None:
        return []
    if not isinstance(document, list):
        raise PlaybookDependencyError(f"Expected a YAML playbook list in {path}")

    imports = []
    for index, item in enumerate(document):
        if not isinstance(item, dict):
            continue
        for key in IMPORT_KEYS:
            if key not in item:
                continue
            import_name = item[key]
            if not isinstance(import_name, str) or not import_name.strip():
                raise PlaybookDependencyError(
                    f"{path}: import_playbook at list item {index} must be a literal filename"
                )
            import_name = import_name.strip()
            if "{{" in import_name or "{%" in import_name:
                raise PlaybookDependencyError(
                    f"{path}: dynamic import paths are not supported: {import_name}"
                )
            imports.append(import_name)
    return imports


def find_dependencies(playbook_name, playbook_root=PLAYBOOK_ROOT):
    """Return the top-level playbook and recursive imports as resolved Paths."""
    root = Path(playbook_root).resolve(strict=True)
    requested = Path(playbook_name)
    if (
        not playbook_name
        or requested.name != playbook_name
        or requested.is_absolute()
        or playbook_name in (".", "..")
    ):
        raise PlaybookDependencyError(
            "playbook must be a filename without directory components"
        )
    if requested.suffix.lower() not in (".yml", ".yaml"):
        raise PlaybookDependencyError("playbook must end in .yml or .yaml")

    top_level = _resolve_playbook(root / requested.name, root)
    ordered_files = []
    visited = set()
    active_stack = []

    def visit(path):
        path = _resolve_playbook(path, root)
        if path in active_stack:
            cycle_start = active_stack.index(path)
            cycle = active_stack[cycle_start:] + [path]
            relative_cycle = " -> ".join(str(item.relative_to(root)) for item in cycle)
            raise PlaybookDependencyError(
                f"Circular import_playbook dependency: {relative_cycle}"
            )
        if path in visited:
            return

        active_stack.append(path)
        ordered_files.append(path)
        for import_name in _read_imports(path):
            import_path = Path(import_name)
            if import_path.is_absolute():
                raise PlaybookDependencyError(
                    f"{path}: absolute import paths are not allowed: {import_name}"
                )
            child_path = _resolve_playbook(path.parent / import_path, root)
            visit(child_path)
        active_stack.pop()
        visited.add(path)

    visit(top_level)
    return ordered_files


def stage_playbooks(playbook_name, job_project_dir, playbook_root=PLAYBOOK_ROOT):
    """Copy validated playbooks, preserving paths relative to the playbook root."""
    root = Path(playbook_root).resolve(strict=True)
    dependency_files = find_dependencies(playbook_name, root)
    project_dir = Path(job_project_dir)
    for source in dependency_files:
        destination = project_dir / source.relative_to(root)
        destination.parent.mkdir(parents=True, exist_ok=True)
        shutil.copy2(source, destination)
    return dependency_files

---
- name: Read-only file and path checks
  hosts: all
  gather_facts: false
  tasks:
    - name: Check that /etc/hostname exists
      ansible.builtin.stat:
        path: /etc/hostname
      register: hostname_file

    - name: Report /etc/hostname metadata
      ansible.builtin.debug:
        msg:
          - "exists={{ hostname_file.stat.exists }}"
          - "isreg={{ hostname_file.stat.isreg | default(false) }}"
          - "size={{ hostname_file.stat.size | default('n/a') }}"

    - name: Read /etc/hostname
      ansible.builtin.slurp:
        src: /etc/hostname
      register: hostname_contents
      when: hostname_file.stat.exists

    - name: Display /etc/hostname contents
      ansible.builtin.debug:
        msg: "{{ hostname_contents.content | b64decode | trim }}"
      when: hostname_file.stat.exists

    - name: Check that /etc/hosts exists
      ansible.builtin.stat:
        path: /etc/hosts
      register: hosts_file

    - name: Report /etc/hosts existence
      ansible.builtin.debug:
        msg: "/etc/hosts exists={{ hosts_file.stat.exists }}"
